### 基于无名杀的架构改进

#### 1. 技能系统增强
```python
class Skill:
    def __init__(self, name, trigger, filter_func, cost_func, content_func):
        self.name = name
        self.trigger = trigger      # 触发时机字典
        self.filter = filter_func   # 条件判断
        self.cost = cost_func       # 成本（询问是否发动）
        self.content = content_func # 效果执行
    
    async def can_trigger(self, event, player):
        """检查是否可以触发"""
        if not self.filter:
            return True
        return await self.filter(event, player)
    
    async def execute(self, event, trigger, player):
        """执行技能"""
        # 1. 询问成本
        if self.cost:
            result = await self.cost(event, trigger, player)
            if not result:
                return False
        
        # 2. 执行效果
        if self.content:
            await self.content(event, trigger, player)
        
        return True
```

#### 2. 事件触发系统
```python
class EventTrigger:
    def __init__(self):
        self.triggers = {}  # {event_name: [skills]}
    
    def register_skill(self, event_name, skill):
        """注册技能到事件"""
        if event_name not in self.triggers:
            self.triggers[event_name] = []
        self.triggers[event_name].append(skill)
    
    async def trigger_event(self, event_name, event_data, player):
        """触发事件，执行所有监听的技能"""
        if event_name not in self.triggers:
            return
        
        for skill in self.triggers[event_name]:
            if await skill.can_trigger(event_data, player):
                await skill.execute(event_data, event_data, player)
```

#### 3. 卡牌使用流程
```python
async def use_card(self, player, card, targets):
    """使用卡牌的完整流程"""
    # 1. 创建使用卡牌事件
    event = {
        'name': 'useCard',
        'player': player,
        'card': card,
        'targets': targets
    }
    
    # 2. 触发"使用前"事件
    await self.trigger_event('useCardBefore', event, player)
    
    # 3. 执行卡牌效果
    if card.name == '杀':
        await self.use_sha(player, targets[0])
    elif card.name == '桃':
        await player.recover(1)
    
    # 4. 触发"使用后"事件
    await self.trigger_event('useCardAfter', event, player)
    
    # 5. 记录历史
    player.history['useCard'].append(event)
```
