#### Phase 3: 阶段控制器

```python
class PhaseController:
    def execute_phase(self, game_state, phase):
        """执行指定阶段"""
        if phase == 'PREPARE':
            self.prepare_phase(game_state)
        elif phase == 'JUDGE':
            self.judge_phase(game_state)
        # ... 其他阶段
    
    def prepare_phase(self, game_state):
        """准备阶段：翻面武将，跳过判定和摸牌"""
        player = game_state.current_player
        if player.is_turned:
            player.turn_over()
            game_state.skip_next_phases(['JUDGE', 'DRAW'])
    
    def draw_phase(self, game_state):
        """摸牌阶段：摸两张牌"""
        player = game_state.current_player
        self.card_manager.draw_card(player, 2)
```

#### Phase 5: 伤害系统
```python
class DamageInfo:
    def __init__(self, source, target, amount, damage_type):
        self.source = source       # 伤害来源
        self.target = target       # 目标
        self.amount = amount       # 伤害点数
        self.type = damage_type    # 伤害类型
        self.prevented = False     # 是否被防止

class DamageSystem:
    def deal_damage(self, damage_info, game_state):
        """造成伤害的完整流程"""
        # 1. 伤害即将造成（技能触发点）
        self.trigger_before_damage(damage_info)
        
        # 2. 防具效果判定
        if self.check_armor(damage_info):
            return
        
        # 3. 造成伤害
        damage_info.target.hp -= damage_info.amount
        
        # 4. 伤害已造成（技能触发点）
        self.trigger_after_damage(damage_info)
        
        # 5. 检查濒死
        if damage_info.target.hp <= 0:
            self.enter_dying(damage_info.target)
```

#### Phase 6: 判定系统
```python
class JudgeSystem:
    def judge(self, player, judge_type):
        """执行判定"""
        # 1. 亮出判定牌
        judge_card = self.deck.pop(0)
        
        # 2. 技能修改判定牌（鬼才、天妒等）
        judge_card = self.trigger_judge_modify(player, judge_card)
        
        # 3. 判定结果
        result = self.evaluate_judge(judge_card, judge_type)
        
        # 4. 判定牌进入弃牌堆
        self.discard_pile.append(judge_card)
        
        return result
```

#### Phase 7: AI决策系统
```python
class AIPlayer:
    def __init__(self, hero, api_config):
        self.hero = hero
        self.api_config = api_config
        
    async def make_decision(self, game_state, available_actions):
        """AI做决策"""
        # 1. 构建prompt（包含当前状态+worldbook）
        prompt = self.build_prompt(game_state)
        
        # 2. 调用AI API
        response = await self.call_ai_api(prompt)
        
        # 3. 解析AI响应
        action = self.parse_ai_response(response)
        
        # 4. 验证合法性
        if action in available_actions:
            return action
        else:
            return self.fallback_decision(available_actions)
```

### 开发流程示例

#### 示例：实现"杀"这张牌

```python
# 1. 定义卡牌
sha_card = Card(
    name='杀',
    suit='spade',
    rank=7,
    type='basic',
    effect=ShaEffect()
)

# 2. 实现效果
class ShaEffect:
    def can_use(self, player, targets, game_state):
        """验证是否可以出"""
        # 检查目标数量
        if len(targets) != 1:
            return False
        
        # 检查距离
        target = targets[0]
        distance = game_state.get_distance(player, target)
        weapon_range = player.get_weapon_range()
        
        if distance > weapon_range:
            return False
        
        # 检查出杀次数（默认每回合1次）
        if player.sha_count >= player.max_sha:
            return False
        
        return True
    
    def execute(self, player, targets, game_state):
        """执行效果"""
        target = targets[0]
        
        # 1. 触发"使用杀"时机的技能
        game_state.trigger_event('USE_SHA', player, target)
        
        # 2. 要求目标响应
        response = game_state.request_response(target, 'SHAN')
        
        # 3. 结算
        if response is None:
            # 没有闪，造成1点伤害
            damage_info = DamageInfo(
                source=player,
                target=target,
                amount=1,
                type='normal'
            )
            game_state.damage_system.deal_damage(damage_info)
        else:
            # 有闪，抵消
            game_state.discard_card(target, response)
        
        # 4. 增加出杀计数
        player.sha_count += 1
```
