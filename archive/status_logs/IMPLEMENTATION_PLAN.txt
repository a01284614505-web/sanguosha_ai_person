```javascript
sha: {

    async content(event) {
        const target = event.target;
        
        // 1. 要求目标打出闪
        const result = await target.chooseToUse("请使用一张闪响应杀");
        
        // 2. 判断是否闪避
        if (isShaned(result)) {
            // 闪避成功

            // 没闪，造成伤害

            trigger("shaHit");
        }
    }
}
```

**核心流程：**
1. 检查距离
2. 要求响应（闪）
3. 判断结果
4. 造成伤害或触发miss

```python
# 1. 完整的技能系统
class Skill:
    def __init__(self, name, config):
        self.name = name
        self.audio = config.get('audio')
        self.trigger = config.get('trigger', {})
        self.filter_func = config.get('filter')
        self.cost_func = config.get('cost')
        self.content_func = config.get('content')
        self.forced = config.get('forced', False)
    
    async def check_trigger(self, event_name, event, player):
        """检查是否应该触发"""
        # 检查触发时机
        if 'player' in self.trigger:
            if self.trigger['player'] != event_name:
                return False
            if event.player != player:
                return False
        
        if 'global' in self.trigger:
            if self.trigger['global'] != event_name:
                return False
        
        # 检查过滤条件
        if self.filter_func:
            return await self.filter_func(event, player)
        
        return True
    
    async def execute(self, event, trigger, player):
        """执行技能"""
        # 1. 询问成本（是否发动）
        if not self.forced and self.cost_func:
            result = await self.cost_func(event, trigger, player)
            if not result or not result.get('bool'):
                return False
        
        # 2. 执行效果
        if self.content_func:
            await self.content_func(event, trigger, player)
        
        return True

# 2. 事件触发管理器
class TriggerManager:
    def __init__(self):
        self.registered_skills = {}
    
    def register_skill(self, player, skill):
        """注册技能"""
        player_id = id(player)
        if player_id not in self.registered_skills:
            self.registered_skills[player_id] = []
        self.registered_skills[player_id].append(skill)
    
    async def trigger(self, event_name, event, players=None):
        """触发事件"""
        if players is None:
            players = self.get_all_players()
        
        for player in players:
            player_id = id(player)
            if player_id not in self.registered_skills:
                continue
            
            for skill in self.registered_skills[player_id]:
                if await skill.check_trigger(event_name, event, player):
                    await skill.execute(event, event, player)

# 3. 卡牌使用系统
class CardSystem:
    async def use_sha(self, player, card, target):
        """使用杀"""
        event = {
            'name': 'sha',
            'player': player,
            'card': card,
            'target': target,
            'baseDamage': 1,
            'shanRequired': 1
        }
        
        # 触发使用前事件
        await self.trigger_manager.trigger('useCard', event)
        
        # 执行杀的效果
        await self._sha_effect(event)
    
    async def _sha_effect(self, event):
        """杀的效果"""
        player = event['player']
        target = event['target']
        
        # 1. 要求打出闪
        result = await self.choose_to_respond(
            target, 
            "请使用一张闪响应杀",
            card_filter=lambda c: c.name == '闪'
        )
        
        # 2. 判断结果
        if result and result.get('bool'):
            # 闪避成功
            await self.trigger_manager.trigger('shaMiss', event)
        else:
            # 造成伤害
            await self.damage(player, target, event['baseDamage'])
            await self.trigger_manager.trigger('shaHit', event)
    
    async def choose_to_respond(self, player, prompt, card_filter):
        """选择响应（人类或AI）"""
        if player.is_ai:
            # AI决策
            return await self.ai_respond(player, prompt, card_filter)
        else:
            # 人类玩家（前端交互）
            return await self.player_respond(player, prompt, card_filter)

# 4. AI决策接口
class AIDecision:
    async def ai_respond(self, player, prompt, card_filter):
        """AI响应决策"""
        # 找到符合条件的牌
        valid_cards = [c for c in player.hand if card_filter(c)]
        
        if not valid_cards:
            return {'bool': False}
        
        # 简单AI：直接打出第一张
        # 未来可以接入LLM做决策
        return {
            'bool': True,
            'card': valid_cards[0]
        }
    
    async def ai_use_card(self, player, available_cards):
        """AI出牌决策"""
        # 这里可以调用AI API
        prompt = f"""
你是{player.hero.name}，当前手牌：{[c.name for c in player.hand]}
场上情况：{self.get_game_state()}
请选择要使用的牌和目标。
"""
        
        # 调用AI API获取决策
        decision = await self.call_ai_api(prompt, player.worldbook)
        
        return decision
```
