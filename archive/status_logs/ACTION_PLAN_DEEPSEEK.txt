### 脚本2：从BWIKI抓取武将数据

```python
#!/usr/bin/env python3
# scrape_wiki.py

import requests
from bs4 import BeautifulSoup
import json

def scrape_hero(url):
    """从BWIKI抓取武将数据"""
    response = requests.get(url)
    soup = BeautifulSoup(response.content, 'html.parser')
    
    # 解析页面（根据实际HTML结构调整）
    hero_data = {
        "name": soup.select_one('.hero-name').text,
        "faction": soup.select_one('.faction').text,
        "hp": int(soup.select_one('.hp').text),
        "skills": []
    }
    
    return hero_data

# 使用
url = "https://wiki.biligame.com/msgs/界曹操"
data = scrape_hero(url)
print(json.dumps(data, ensure_ascii=False, indent=2))
```

### 脚本3：从无名杀JS提取数据

```javascript
// extract_from_noname.js
const fs = require('fs');

// 读取无名杀的武将文件
const heroFile = fs.readFileSync('./noname/character/standard.js', 'utf8');

// 简单的正则提取（实际可能需要更复杂的解析）
const heroes = {};

// 提取武将定义
const heroPattern = /(\w+):\s*\[(['"].*?['"])\]/g;
let match;

while ((match = heroPattern.exec(heroFile)) !== null) {
    const heroId = match[1];
    const heroData = match[2];
    heroes[heroId] = heroData;
}

fs.writeFileSync('extracted_heroes.json', JSON.stringify(heroes, null, 2));
console.log('✓ 已提取', Object.keys(heroes).length, '个武将');
```
