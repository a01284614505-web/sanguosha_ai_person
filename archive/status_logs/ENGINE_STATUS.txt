```python
from rules_engine import RulesEngine

class MainEngine:
    def __init__(self, config):
        # ... 现有代码 ...
        self.rules = RulesEngine(self.game_state)
        self.ai_decision = AIDecision(self)
    
    async def execute_play_phase(self, player):
        """出牌阶段"""
        while True:
            # 获取可选操作
            actions = self.get_available_actions(player)
            
            # 如果是AI
            if player.is_ai:
                decision = await self.ai_decision.make_decision(player, actions)
                action = decision['action']
            else:
                # 等待玩家操作（WebSocket）
                action = await self.wait_player_action(player, actions)
            
            # 执行操作
            if action['type'] == 'end_phase':
                break
            
            await self.execute_action(player, action)
```
