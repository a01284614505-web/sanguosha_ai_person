### 方案2：注入AI世界书

```python
def build_ai_worldbook(game_state):
    """构建当前局面的世界书"""
    worldbook = []
    
    # 1. 注入游戏规则
    with open('worldbook_generated/rules/game_flow.json') as f:
        rules = json.load(f)
        current_phase = game_state.current_phase
        phase_info = next(p for p in rules['phases'] if p['id'] == current_phase)
        worldbook.append(f"【当前阶段】{phase_info['description']}")
    
    # 2. 注入在场武将技能
    with open('worldbook_generated/heroes/heroes_27_complete.json') as f:
        all_heroes = json.load(f)['heroes']
        for player in game_state.players:
            hero = next(h for h in all_heroes if h['id'] == player.hero_id)
            for skill in hero['skills']:
                worldbook.append(f"【{hero['name']}·{skill['name']}】{skill['worldbook']}")
    
    # 3. 注入手牌信息（仅对当前玩家）
    if game_state.current_player.is_ai:
        with open('worldbook_generated/cards/basic_cards.json') as f:
            cards_db = json.load(f)['cards']
            hand_cards = game_state.current_player.hand
            for card in hand_cards:
                card_info = next(c for c in cards_db if c['id'] == card.id)
                worldbook.append(f"【{card_info['name']}】{card_info.get('worldbook', '')}")
    
    return "\n\n".join(worldbook)

# 在AI决策时调用
ai_prompt = f"""
你是【{player.hero.name}】，身份：{player.identity}

{build_ai_worldbook(game_state)}

当前局势：
- 你的体力：{player.hp}/{player.max_hp}
- 你的手牌：{len(player.hand)}张
- 场上存活：{len([p for p in players if p.alive])}人

可选操作：
{format_available_actions(actions)}

请从以上操作中选择一个，返回JSON格式。
"""
```

def build_decision_prompt(self, player, available_actions):
    prompt = f"你是【{player.hero.name}】\n\n"
    
    # 注入武将技能世界书
    hero_wb = self.load_hero_worldbook(player.hero_id)
    prompt += f"你的技能：\n{hero_wb}\n\n"
    
    # 注入当前阶段规则
    phase_wb = self.load_phase_worldbook(self.engine.game_state.current_phase)
    prompt += f"当前阶段：\n{phase_wb}\n\n"
    
    # 注入可选操作
    prompt += f"可选操作：\n{format_actions(available_actions)}"
    
    return prompt

def load_hero_worldbook(self, hero_id):
    """加载武将世界书"""
    with open('data/worldbook/heroes/heroes_27_complete.json') as f:
        heroes = json.load(f)['heroes']

        worldbook = []
        for skill in hero['skills']:
            worldbook.append(f"【{skill['name']}】{skill['worldbook']}")
        
        return "\n".join(worldbook)

## 🎯 高级用法

### 1. 动态世界书注入（按需加载）

```python
class WorldbookManager:
    def __init__(self):
        self.cache = {}
    
    def get_relevant_worldbook(self, game_state):
        """只注入相关的世界书条目"""
        worldbook = []
        
        # 1. 当前阶段规则
        worldbook.append(self.get_phase_info(game_state.current_phase))
        
        # 2. 在场武将技能
        for player in game_state.players:
            if player.alive:
                worldbook.append(self.get_hero_skills(player.hero_id))
        
        # 3. 手牌说明（只加载手中有的牌）
        current_player = game_state.current_player
        for card in current_player.hand:
            worldbook.append(self.get_card_info(card.id))
        
        return "\n\n".join(worldbook)
```

### 2. Token预算控制

```python
def build_worldbook_with_budget(game_state, max_tokens=2000):
    """控制世界书不超过token预算"""
    worldbook = []
    current_tokens = 0
    
    # 优先级：当前阶段 > 自己技能 > 敌方技能 > 手牌
    priorities = [
        ('phase', get_phase_info()),
        ('self_skills', get_self_hero_skills()),
        ('enemy_skills', get_enemy_skills()),
        ('cards', get_hand_cards_info())
    ]
    
    for priority, content in priorities:
        content_tokens = estimate_tokens(content)
        if current_tokens + content_tokens <= max_tokens:
            worldbook.append(content)
            current_tokens += content_tokens
        else:
            break  # 预算用完
    
    return "\n\n".join(worldbook)
```

## 🐛 常见问题

### Q1: JSON文件很大，加载慢怎么办？

**A**: 按需加载，缓存已加载的数据

```python
class WorldbookCache:
    _cache = {}
    
    @classmethod
    def load_hero(cls, hero_id):
        if hero_id not in cls._cache:
            # 第一次加载时缓存
            hero = load_from_json(hero_id)
            cls._cache[hero_id] = hero
        return cls._cache[hero_id]
```

### Q2: 世界书太长，AI token超限？

**A**: 使用优先级裁剪（见上面"Token预算控制"）
